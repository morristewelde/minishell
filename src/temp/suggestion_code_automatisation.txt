int	builtin_cd(t_command *cmd, t_shell *shell)
{
	char	*path;
	char	cwd[1024];
	char	*old_pwd;
	int		from_cd_dash = 0; // Flag pour indiquer si "cd -" ou "cd ." est utilisé

	// Obtenir le chemin actuel (PWD)
	if (!getcwd(cwd, sizeof(cwd)))
	{
		perror("minishell: cd: getcwd");
		shell->last_exit_status = 1; // Mettre à jour le statut de sortie
		return (1);
	}

	// Si aucun argument, utiliser HOME
	if (!cmd->argv[1])
	{
		path = get_env_value(shell->env, "HOME");
		if (!path || !*path)
		{
			ft_putstr_fd("minishell: cd: HOME not set\n", STDERR_FILENO);
			shell->last_exit_status = 1; // Mettre à jour le statut de sortie
			return (1);
		}
		path = ft_strdup(path);
	}
	// Si "cd -" utilisé, alterner entre OLDPWD et PWD
	else if (ft_strcmp(cmd->argv[1], "-") == 0)
	{
		path = get_env_value(shell->env, "OLDPWD");
		if (!path || !*path)
		{
			ft_putstr_fd("minishell: cd: OLDPWD not set\n", STDERR_FILENO);
			shell->last_exit_status = 1; // Mettre à jour le statut de sortie
			return (1);
		}
		path = ft_strdup(path);
		from_cd_dash = 1;

		// Affiche OLDPWD immédiatement (spécifique à cd -)
		ft_putstr_fd(path, STDOUT_FILENO);
		ft_putstr_fd("\n", STDOUT_FILENO);
	}
	// Si "cd ." utilisé, marquer comme déjà affiché
	else if (ft_strcmp(cmd->argv[1], ".") == 0)
	{
		from_cd_dash = 1; // Indique qu'on ne doit rien afficher
		path = ft_strdup(cwd); // Le chemin reste le même
	}
	// Sinon, utiliser le chemin donné
	else
		path = ft_strdup(cmd->argv[1]);

	// Vérifie si le chemin existe
	if (access(path, F_OK) != 0)
	{
		ft_putstr_fd("minishell: cd: ", STDERR_FILENO);
		ft_putstr_fd(path, STDERR_FILENO);
		ft_putstr_fd(": No such file or directory\n", STDERR_FILENO);
		free(path);
		shell->last_exit_status = 1; // Mettre à jour le statut de sortie
		return (1);
	}

	// Changer de répertoire
	if (chdir(path) == -1)
	{
		perror("minishell: cd");
		free(path);
		shell->last_exit_status = 1; // Mettre à jour le statut de sortie
		return (1);
	}

	// Mettre à jour OLDPWD et PWD
	old_pwd = get_env_value(shell->env, "PWD");
	if (old_pwd)
		set_env_var(&shell->env, "OLDPWD", old_pwd);
	if (getcwd(cwd, sizeof(cwd)))
		set_env_var(&shell->env, "PWD", cwd);
	else
		perror("minishell: cd: getcwd");

	// Affiche le chemin si ce n’est pas déjà fait via "cd -" ou "cd ."
	if (!from_cd_dash)
	{
		ft_putstr_fd(cwd, STDOUT_FILENO);
		ft_putstr_fd("\n", STDOUT_FILENO);
	}

	free(path);
	shell->last_exit_status = 0; // Mettre à jour le statut de sortie en cas de succès
	return (0);
}

void	builtin_echo(char **args, t_shell *shell)
{
	int	i;
	int	newline;

	printf("[DEBUG] builtin_echo() called with args:\n");
	for (i = 0; args[i]; i++)
		printf("[DEBUG] Arg[%d] = %s\n", i, args[i]);

	i = 1;
	newline = 1;

	// Vérifie si -n est activé
	while (args[i] && is_valid_n_option(args[i]))
	{
		newline = 0;
		i++;
	}

	// Affichage des arguments avec gestion correcte des espaces
	while (args[i])
	{
		if (ft_putstr_fd(args[i], STDOUT_FILENO) < 0)
		{
			perror("minishell: echo");
			shell->last_exit_status = 1; // Mettre à jour le statut de sortie en cas d'erreur
			return;
		}
		if (args[i + 1] && ft_putstr_fd(" ", STDOUT_FILENO) < 0)
		{
			perror("minishell: echo");
			shell->last_exit_status = 1; // Mettre à jour le statut de sortie en cas d'erreur
			return;
		}
		i++;
	}

	if (!newline)
	{
		if (ft_putstr_fd(" \033[0K", STDOUT_FILENO) < 0)
		{
			perror("minishell: echo");
			shell->last_exit_status = 1; // Mettre à jour le statut de sortie en cas d'erreur
			return;
		}
	}

	if (newline && ft_putstr_fd("\n", STDOUT_FILENO) < 0)
	{
		perror("minishell: echo");
		shell->last_exit_status = 1; // Mettre à jour le statut de sortie en cas d'erreur
		return;
	}

	shell->last_exit_status = 0; // Mettre à jour le statut de sortie en cas de succès

	printf("[DEBUG] builtin_echo() finished execution\n");
}

int	validate_commands(t_command *commands, t_shell *shell)
{
	t_command	*cmd = commands;

	while (cmd)
	{
		if (!cmd->argv || !cmd->argv[0])
		{
			ft_putstr_fd("minishell: syntax error: unexpected empty command\n", STDERR_FILENO);
			shell->last_exit_status = 1; // Mettre à jour le statut de sortie

et realloc comme fonction safe realloc?

Absolument ! Une fonction safe_realloc est une excellente idée pour gérer les réallocations de mémoire de manière sécurisée et centralisée. Elle permet de réallouer de la mémoire tout en vérifiant les erreurs et en évitant les fuites de mémoire.

Voici comment tu peux implémenter safe_realloc :
Implémentation de safe_realloc
c
Copy

void *safe_realloc(void *ptr, size_t size)
{
    void *new_ptr;

    new_ptr = realloc(ptr, size);
    if (!new_ptr && size > 0) // Vérifie si la réallocation a échoué
    {
        print_error("realloc", "memory allocation failed");
        free(ptr); // Libère l'ancien pointeur pour éviter une fuite de mémoire
        exit(EXIT_FAILURE);
    }
    return (new_ptr);
}

Explications :

    realloc(ptr, size) : Réalloue la mémoire pointée par ptr à la nouvelle taille size.

    Vérification des erreurs : Si realloc échoue, la fonction affiche une erreur et quitte le programme.

    Libération de l'ancien pointeur : Si realloc échoue, free(ptr) est appelé pour éviter une fuite de mémoire.

    Retour du nouveau pointeur : Si la réallocation réussit, la fonction retourne le nouveau pointeur.

et comme jai pas droit a la fonction realloc c possible de la creer?

Ah, je vois ! Si tu n'as pas le droit d'utiliser la fonction realloc directement, tu peux la recréer manuellement en utilisant malloc, free, et memcpy. Cela te permettra de gérer la réallocation de mémoire tout en respectant les contraintes de ton projet.

Voici comment tu peux implémenter ta propre version de realloc, que nous appellerons ft_realloc, ainsi qu'une version sécurisée safe_ft_realloc.
Implémentation de ft_realloc
Fonctionnement :

    Si ptr est NULL, ft_realloc se comporte comme malloc.

    Si size est 0, ft_realloc se comporte comme free.

    Sinon, ft_realloc alloue un nouveau bloc de mémoire, copie les données de l'ancien bloc vers le nouveau, et libère l'ancien bloc.

Code :
c
Copy

#include <stdlib.h>
#include <string.h> // Pour memcpy

void *ft_realloc(void *ptr, size_t old_size, size_t new_size)
{
    void *new_ptr;

    // Si new_size est 0, libère le bloc de mémoire
    if (new_size == 0)
    {
        free(ptr);
        return (NULL);
    }

    // Si ptr est NULL, alloue un nouveau bloc
    if (!ptr)
        return (malloc(new_size));

    // Alloue un nouveau bloc de mémoire
    new_ptr = malloc(new_size);
    if (!new_ptr)
        return (NULL);

    // Copie les données de l'ancien bloc vers le nouveau
    size_t copy_size = (old_size < new_size) ? old_size : new_size;
    memcpy(new_ptr, ptr, copy_size);

    // Libère l'ancien bloc
    free(ptr);

    return (new_ptr);
}
