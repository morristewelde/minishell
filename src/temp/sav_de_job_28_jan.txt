char *format_input(const char *input, t_env_var *env)
{
    char *formatted_input;
    int i = 0;
    int quote_state = NO_QUOTE;

    formatted_input = malloc(1);
    if (!formatted_input)
        return (NULL);
    formatted_input[0] = '\0';

    while (input[i])
    {
        // Gestion des quotes
        if (input[i] == '\'' || input[i] == '"')
            i = handle_quotes_segment(input, i, env, &formatted_input, &quote_state);

        // Gestion des backslashes
        else if (input[i] == '\\')
            i = handle_backslash_segment(input, i, &formatted_input, quote_state);

        // Gestion des expansions de variables
        else if (input[i] == '$' && quote_state != SINGLE_QUOTE)
            i = handle_expansion_segment(input, i, env, &formatted_input);

        // Gestion des redirections (`>`, `<`, `>>`, `<<`)
        else if (is_redirection(input[i]))
            i = handle_redirection_segment(input, i, &formatted_input);

        // Gestion des pipes
        else if (input[i] == '|')
            i = handle_pipe_segment(input, i, &formatted_input);

        // Gestion des caractères normaux (mots simples)
        else
            i = handle_word_segment(input, i, &formatted_input, env);
    }

    return formatted_input;
}

3. Gestion des Backslashes (handle_backslash_segment)

On doit gérer correctement le backslash selon qu’il est dans des quotes simples, doubles ou hors quotes.

int handle_backslash_segment(const char *input, int i, char **formatted_input, int quote_state)
{
    if (quote_state == SINGLE_QUOTE)
    {
        // Dans les quotes simples, le backslash est pris tel quel
        *formatted_input = append_char(*formatted_input, input[i]);
    }
    else if (quote_state == DOUBLE_QUOTE)
    {
        // Dans les quotes doubles, il échappe uniquement "\", "$" et '"'
        if (input[i + 1] == '\\' || input[i + 1] == '$' || input[i + 1] == '"')
            i++; // Ignore le backslash et ajoute le caractère suivant
        *formatted_input = append_char(*formatted_input, input[i]);
    }
    else
    {
        // Hors quotes, on prend simplement le caractère suivant
        if (input[i + 1] != '\0')
            i++;
        *formatted_input = append_char(*formatted_input, input[i]);
    }

    return i + 1;
}

4. Gestion des Quotes (handle_quotes_segment)

    Les quotes simples empêchent toute expansion et prennent tout tel quel.
    Les quotes doubles permettent l'expansion des variables ($VAR).

int handle_quotes_segment(const char *input, int i, t_env_var *env, char **formatted_input, int *quote_state)
{
    char quote_char = input[i]; // Quote ouvrante (' ou ")
    int start = ++i; // Avancer après la quote ouvrante

    *quote_state = (quote_char == '\'') ? SINGLE_QUOTE : DOUBLE_QUOTE;

    while (input[i] && input[i] != quote_char)
        i++;

    if (input[i] != quote_char)
    {
        fprintf(stderr, "[ERROR] Unclosed quote starting at index %d\n", start - 1);
        return -1;
    }

    char *quote_content = ft_substr(input, start, i - start);
    if (!quote_content)
        return -1;

    char *processed_content;
    if (*quote_state == DOUBLE_QUOTE)
        processed_content = replace_env_variables(quote_content, env);
    else
        processed_content = ft_strdup(quote_content);

    free(quote_content);
    if (!processed_content)
        return -1;

    char *temp = ft_strjoin(*formatted_input, processed_content);
    free(*formatted_input);
    free(processed_content);
    *formatted_input = temp;

    *quote_state = NO_QUOTE;
    return i + 1;
}

5. Gestion des Expansions (handle_expansion_segment)

On remplace $VAR par sa valeur si elle existe, sinon on supprime $VAR.

int handle_expansion_segment(const char *input, int i, t_env_var *env, char **formatted_input)
{
    int var_start = ++i;
    while (isalnum(input[i]) || input[i] == '_')
        i++;

    char *var_name = ft_substr(input, var_start, i - var_start);
    if (!var_name)
        return -1;

    t_env_var *env_var = env_find_var(env, var_name);
    char *var_value = env_var ? ft_strdup(env_var->value) : ft_strdup("");
    free(var_name);
    if (!var_value)
        return -1;

    char *temp = ft_strjoin(*formatted_input, var_value);
    free(*formatted_input);
    free(var_value);
    *formatted_input = temp;

    return i;
}

6. Gestion des Redirections (handle_redirection_segment)

On détecte >, <, >>, << et les sépare correctement des mots suivants.

int handle_redirection_segment(const char *input, int i, char **formatted_input)
{
    *formatted_input = append_char(*formatted_input, input[i]);

    if ((input[i] == '>' && input[i + 1] == '>') || (input[i] == '<' && input[i + 1] == '<'))
    {
        *formatted_input = append_char(*formatted_input, input[i + 1]);
        i++;
    }

    return i + 1;
}

7. Gestion des Pipes (handle_pipe_segment)

On s’assure que | est bien entouré de commandes valides.

int handle_pipe_segment(const char *input, int i, char **formatted_input)
{
    if (i == 0 || input[i + 1] == '|')
    {
        fprintf(stderr, "[ERROR] Syntax error near unexpected token `|`\n");
        return -1;
    }

    *formatted_input = append_char(*formatted_input, '|');
    return i + 1;
}

8. Gestion des Mots (handle_word_segment)

On prend un mot jusqu’à un espace, un pipe, une redirection, ou une quote.

int handle_word_segment(const char *input, int i, char **formatted_input, t_env_var *env)
{
    while (input[i] && !isspace(input[i]) && input[i] != '|' && !is_redirection(input[i])
        && input[i] != '\'' && input[i] != '"')
    {
        *formatted_input = append_char(*formatted_input, input[i]);
        i++;
    }
    return i;
}

Pour avoir un comportement plus proche d’un vrai shell vis-à-vis des backslash et des quotes, il faut clarifier la logique souhaitée. Actuellement, ton code mélange plusieurs règles d’échappement ; par exemple :

    En double quotes : on veut que \" devienne " (donc le backslash disparaît), idem pour \$ ou \\.
    En single quotes : on veut tout prendre littéralement (le backslash n’a pas de rôle d’échappement).
    Hors quotes : on veut autoriser \ à échapper certains caractères (par exemple $, ', ", \) et ignorer le backslash sinon.

Tes tests montrent des divergences entre le code actuel et le “désired behavior” :

    "echo "text with \"escaped quotes\" inside"
        Désiré : echo text with "escaped quotes" inside
        Actuel : echo text with \escaped quotes\ inside

    Le parser garde les \ autour de escaped quotes. Tu voudrais qu’ils disparaissent dans le résultat final pour obtenir des guillemets normaux.

    "echo trailing\\ backslash"
        Désiré : echo trailing\ backslash (un seul \)
        Actuel : echo trailing\\ backslash (deux \)

    Ici, le parser n’a pas “fusionné” les deux \\ en un seul.

    "echo test\ value"
        Désiré : echo test value (le backslash doit disparaître)
        Actuel : echo test\ value (le backslash reste)

    "echo "A \"quoted\" word"
        Désiré : echo A "quoted" word
        Actuel : echo A \quoted\ word

    Là aussi, le parser garde les \ au lieu de les utiliser pour échapper ".

En gros, ton code n’applique pas la même logique que le shell pour les backslash. Il supprime parfois le \, parfois non, etc.
Rappels de la logique d’un shell (simpliﬁé)
1) Single quotes ( '...' )

    Tout est littéral dedans, sauf la quote fermante '.
    Le backslash n’a aucun effet spécial.
        Par exemple, 'abc\def' => littéralement abc\def.

2) Double quotes ( "..." )

    On autorise l’expansion de $ (et \ sert à échapper certains caractères).
    Le backslash dans les double quotes sert à échapper ", \, et $.
        \" => un guillemet " dans la chaîne finale,
        \\ => un seul backslash,
        \$ => un $ littéral, au lieu de commencer une expansion.
    Les autres backslash (ex. \a) ne sont pas spéciaux dans un shell posix standard (même si Bash gère un peu plus).
        Du coup, echo "A \foo" => A \foo (le backslash reste).

3) Hors quotes

    \ échappe certains caractères spéciaux : $, ', ", \, parfois (espace) selon le shell…
    Si \ précède un caractère non spécial, alors le backslash disparait ou reste ? (En bash, \c => c).

Comment adapter ton code ?

    Dans handle_backslash_segment(), il faut différencier :
        Hors quotes (quote_state == NO_QUOTE) :
            Si input[i+1] est un caractère “spécial” ($, ', ", \, peut-être un espace), on saute le \ et on ajoute le caractère suivant.
            Sinon, on retire le backslash et on ajoute le caractère suivant.
        En single quotes :
            Le backslash n’a pas d’effet particulier -> on le recopie tel quel.
        En double quotes :
            Si input[i+1] est " ou $ ou \, on saute le \ et on ajoute l’autre.
            Sinon, on laisse le backslash tel quel (ou on l’enlève, selon la compatibilité POSIX/Bash).

    Ne pas confondre l’échappement du backslash avec d’autres substitutions.
        Si tu as par exemple \ hors quotes devant un $, tu veux ignorer l’expansion.
        Dans tes tests, echo test\ value => on veut echo test value (le backslash fait disparaître l’espace ou le v ?). Normalement, en shell, test\ value => test value.
            Donc, hors quotes, \ (backslash + espace) => un espace littéral.
            Actuellement, ton code laisse test\ value. Il faut l’interpréter en “un escape sur l’espace”.

    Dans replace_env_variables() ou la logique de $, tu fais déjà un bon travail, mais assure-toi de ne pas re-expanser si tu as déjà consommé un backslash. Par exemple, si c’est \$ dans les double quotes et qu’on a sauté le backslash, on veut un $ littéral, pas une expansion.

    Pour “Escaped "quote" test”
        Ton code renvoie Escaped \quote\ test, alors que tu voulais (null) (selon tes logs ?).
        Peut-être que tu voulais lever une erreur si l’utilisateur tape \" hors quotes ? Ou si c’est mal formé ?
        Dans un shell normal, \"quote\" hors quotes va être interprété comme "quote", c’est-à-dire le \ échappe le " suivant.
        Si tu voulais que \"quote\" soit illégal hors quotes, il faut décider d’une règle stricte.
        Bref, tu dois clarifier la règle : “est-ce qu’on accepte \" hors quotes pour mettre un guillemet ?” (Shell normal => oui).

    {
        *formatted = append_char(*formatted, '\\');
        return i + 1;
    }

    // 2) En double quotes
    if (quote_state == DOUBLE_QUOTE)
    {
        if (input[i+1] == '"' || input[i+1] == '$' || input[i+1] == '\\')
        {
            // On saute le '\' et on ajoute input[i+1]
            *formatted = append_char(*formatted, input[i+1]);
            return i + 2; // on avance de 2
        }
        else
        {
            // On copie juste le '\' (ou on l'ignore selon la compat)
            *formatted = append_char(*formatted, '\\');
            return i + 1;
        }
    }

    // 3) Hors quotes (NO_QUOTE)
    {
        // Dans un shell standard, '\' échappe $, ', ", \, et parfois espace
        char c = input[i+1];
        if (c == '$' || c == '\'' || c == '"' || c == '\\' || c == ' ')
        {
            // On saute le '\' et ajoute c
            *formatted = append_char(*formatted, c);
            return i + 2;
        }
        else
        {
            // On supprime le '\' et on ajoute c tel quel
            *formatted = append_char(*formatted, c);
            return i + 2;
        }
    }
}

C’est un schéma plus proche d’un shell. Il faut ensuite tester chaque cas (\", \\, etc.) dans les contextes singles/doubles quotes ou hors quotes.

    Tester les cas :

        Single quotes : echo 'A \'quoted\' word' -> littéral A \quoted` word`.

char	*replace_env_variables(const char *input, t_shell *shell)
{
	char	*result;
	int		i;

	result = malloc(1);
	if (!result)
		return (NULL);
	result[0] = '\0';
	i = 0;
	while (input[i])
	{
		if (input[i] == '$' && (ft_isalnum(input[i + 1]) || input[i + 1] == '_' || input[i + 1] == '?'))
		{
			result = replace_variable(input, &i, result, shell);
			if (!result)
				return (NULL);
		}
		else
		{
			result = append_char(result, input[i]);
			if (!result)
				return (NULL);
			i++;
		}
	}
	return (result);
}

char	*replace_variable(const char *input, int *i, char *result, t_shell *shell)
{
	int		var_start;
	char	*var_name;
	char	*var_value;
	char	*temp;

	(*i)++;
	// 📌 **Cas spécial pour `$?`**
	if (input[*i] == '?')
	{
		var_value = ft_itoa(shell->last_exit_status);
		(*i)++; // Avancer après `?`
	}
	else
	{
		var_start = *i;
		while (ft_isalnum(input[*i]) || input[*i] == '_')
			(*i)++;
		var_name = ft_substr(input, var_start, *i - var_start);
		if (!var_name)
			return (NULL);
		var_value = get_env_value(shell->env, var_name);
		free(var_name);
	}
	if (!var_value)
		var_value = ft_strdup("");

	temp = ft_strjoin(result, var_value);
	free(result);
	free(var_value);
	return (temp);
}


char	*append_char(char *result, char c)
{
	char	temp[2];
	char	*new_result;

	temp[0] = c;
	temp[1] = '\0';
	//printf("[DEBUG] append_char: Appending character '%c' to result\n", c);

	new_result = ft_strjoin(result, temp);
	free(result);
	if (!new_result)
		printf("[DEBUG] append_char: Failed to append character\n");
	return (new_result);
}


char	*process_quotes_and_expansion(const char *input, t_shell *shell)
{
	char			*result;
	char			*sub;
	int				i;
	int				start;
	t_quote_type	quote_state;

	result = malloc(1);
	if (!result)
		return (NULL);
	result[0] = '\0';
	i = 0;
	quote_state = NO_QUOTE;
	while (input[i])
	{
		if (input[i] == '\'' || input[i] == '"')
		{
			quote_state = set_in_quotes(input[i], quote_state);
			start = ++i;
			while (input[i] && set_in_quotes(input[i], quote_state) == quote_state)
				i++;
			if (!input[i])
			{
				free(result);
				return (NULL);
			}
			sub = ft_substr(input, start, i - start);
			if (!sub)
			{
				free(result);
				return (NULL);
			}
			if (quote_state == SINGLE_QUOTE)
				result = ft_strjoin_multiple(2, result, sub);
			else if (quote_state == DOUBLE_QUOTE)
				result = ft_strjoin_multiple(2, result, replace_env_variables(sub, shell));
			free(sub);
			quote_state = set_in_quotes(input[i++], quote_state);
		}
		else if (input[i] == '$')
		{
			start = i;
			while (input[++i] && (ft_isalnum(input[i]) || input[i] == '_'))
				;
			sub = ft_substr(input, start, i - start);
			sub = replace_env_variables(sub, shell);
			if (!sub)
			{
				free(result);
				return (NULL);
			}
			result = ft_strjoin_multiple(2, result, sub);
			free(sub);
		}
		else
		{
			char temp[2] = {input[i++], '\0'};
			result = ft_strjoin_multiple(2, result, temp);
		}
		if (!result)
			return (NULL);
	}
	return (result);
}
